\chapter{Follow-Up Interview Guide}

\section*{Spørsmål til Oppfølgningsintervju}

\textbf{Hva synes du om verktøyet?}
\begin{itemize}
\item Hvor lang tid brukte du det?
\item Hvor enkelt var det å bruke det?
\item Hvilke funksjoner brukte du mest?
\item Hva likte du best med det?
\item Hva likte du minst med det?
\item Hva kan forbedres med det?
\item Hvilke eksterne ressurser brukte du?
\end{itemize}

\textbf{Har du fått nye meninger om språkmodeller?}
\begin{itemize}
\item Har forståelsen din av hvordan språkmodeller fungerer endret seg etter å ha brukt verktøyet?
\item Hva tenker du nå om\ldots?
\begin{itemize}
\item Spør spesifikt om misforståelser de hadde før
\end{itemize}
\end{itemize}

\textbf{Har du blitt mer eller mindre skeptisk til teknologien etter å ha brukt verktøyet?}

\textbf{Kommer du til å forandre bruk av språkmodeller etter å ha brukt verktøyet?}
\begin{itemize}
\item I så fall, hvordan?
\end{itemize}
